cmake_minimum_required(VERSION 3.25)

# Set architectures before project() so the mod, bindings and SDK agree.
if(APPLE AND NOT DEFINED CMAKE_OSX_ARCHITECTURES)
    set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
endif()
project(EditorWaveform VERSION 1.0.1 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_library(kissfft STATIC third_party/kissfft/kiss_fft.c third_party/kissfft/kiss_fftr.c)
set_target_properties(kissfft PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(kissfft PUBLIC third_party/kissfft)
add_library(spectrogram_core STATIC src/spectrogram.cpp)
set_target_properties(spectrogram_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(spectrogram_core PUBLIC src)
target_link_libraries(spectrogram_core PRIVATE kissfft)

option(EDITOR_WAVEFORM_TESTS_ONLY "Build audio/timing tests without the SDK" OFF)
option(BUILD_TESTING "Build waveform tests" ${EDITOR_WAVEFORM_TESTS_ONLY})
include(CTest)
option(EDITOR_WAVEFORM_FMOD_TESTS "Test audio against a local macOS FMOD runtime" OFF)
set(EDITOR_WAVEFORM_FMOD_LIBRARY "" CACHE FILEPATH "Unmodified FMOD runtime from GD's Resources/fmod/lib")
if(BUILD_TESTING)
    add_executable(waveform_tests tests/waveform_tests.cpp)
    target_include_directories(waveform_tests PRIVATE src)
    add_test(NAME waveform_tests COMMAND waveform_tests)
    add_executable(spectrogram_tests tests/spectrogram_tests.cpp)
    target_link_libraries(spectrogram_tests PRIVATE spectrogram_core)
    add_test(NAME spectrogram_tests COMMAND spectrogram_tests)
    if(EDITOR_WAVEFORM_FMOD_TESTS)
        if(NOT APPLE OR NOT DEFINED ENV{GEODE_SDK} OR NOT EXISTS "${EDITOR_WAVEFORM_FMOD_LIBRARY}")
            message(FATAL_ERROR "FMOD tests require macOS, GEODE_SDK and EDITOR_WAVEFORM_FMOD_LIBRARY")
        endif()
        add_executable(waveform_fmod_tests tests/fmod_tests.cpp src/waveform.cpp)
        target_compile_definitions(waveform_fmod_tests PRIVATE WAVEFORM_AUDIO_ONLY)
        target_include_directories(waveform_fmod_tests PRIVATE src "$ENV{GEODE_SDK}/loader/include")
        target_link_libraries(waveform_fmod_tests PRIVATE spectrogram_core "${EDITOR_WAVEFORM_FMOD_LIBRARY}")
        add_test(NAME waveform_fmod_tests COMMAND waveform_fmod_tests)
    endif()
endif()
if(EDITOR_WAVEFORM_TESTS_ONLY)
    return()
endif()

if(NOT DEFINED ENV{GEODE_SDK})
    message(FATAL_ERROR "Set GEODE_SDK to your installed Geode SDK")
endif()
add_subdirectory("$ENV{GEODE_SDK}" "${CMAKE_CURRENT_BINARY_DIR}/geode")
add_library(${PROJECT_NAME} SHARED src/main.cpp src/waveform.cpp src/timeline.cpp src/settings_popup.cpp)
target_link_libraries(${PROJECT_NAME} spectrogram_core)

# Optional local development override: use the API shipped with installed Eclipse.
set(EDITOR_WAVEFORM_ECLIPSE_API "" CACHE PATH "Local Eclipse include directory")
if(EDITOR_WAVEFORM_ECLIPSE_API)
    if(NOT EXISTS "${EDITOR_WAVEFORM_ECLIPSE_API}/eclipse.hpp")
        message(FATAL_ERROR "EDITOR_WAVEFORM_ECLIPSE_API must contain eclipse.hpp")
    endif()
    target_include_directories(${PROJECT_NAME} PRIVATE "${EDITOR_WAVEFORM_ECLIPSE_API}")
    setup_geode_mod(${PROJECT_NAME} EXTERNALS eclipse.eclipse-menu:1.9.4)
else()
    setup_geode_mod(${PROJECT_NAME})
    target_include_directories(${PROJECT_NAME} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/geode-deps/eclipse.eclipse-menu/include")
endif()

# Eclipse's event API is header-only. Older CLIs can still mark its cached
# dependency as required; linking its binary would defeat optional loading.
get_target_property(WAVEFORM_LINK_LIBRARIES ${PROJECT_NAME} LINK_LIBRARIES)
list(FILTER WAVEFORM_LINK_LIBRARIES EXCLUDE REGEX "eclipse\\.eclipse-menu")
set_property(TARGET ${PROJECT_NAME} PROPERTY LINK_LIBRARIES "${WAVEFORM_LINK_LIBRARIES}")
